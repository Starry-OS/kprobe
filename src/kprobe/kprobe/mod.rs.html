10/// The kprobe structure for the current architecture.
11pub type Kprobe<L, F> = Probe<L, F>;
12
13pub(crate) fn __register_kprobe<L: RawMutex + 'static, F: KprobeAuxiliaryOps>(
14    kprobe_point_list: &mut ProbePointList<F>,
15    kprobe_builder: ProbeBuilder<F>,
16) -> Kprobe<L, F> {
17    let address = kprobe_builder.probe_addr();
18    let existed_point = kprobe_point_list.get(&address).map(Clone::clone);
19
20    match existed_point {
21        Some(existed_point) => kprobe_builder.with_probe_point(existed_point).install().0,
22        None => {
23            let (kprobe, probe_point) = kprobe_builder.install();
24            kprobe_point_list.insert(address, probe_point);
25            kprobe
26        }
27    }
28}

30/// Register a kprobe.
31///
32/// # Parameters
33/// - `manager`: The kprobe manager.
34/// - `kprobe_point_list`: The list of kprobe points.
35/// - `kprobe_builder`: The kprobe builder.
36///
37/// # Returns
38/// - An registered kprobe.
39///
40pub fn register_kprobe<L: RawMutex + 'static, F: KprobeAuxiliaryOps>(
41    manager: &mut ProbeManager<L, F>,
42    kprobe_point_list: &mut ProbePointList<F>,
43    kprobe_builder: ProbeBuilder<F>,
44) -> Arc<Kprobe<L, F>> {
45    let kprobe = __register_kprobe(kprobe_point_list, kprobe_builder);
46    let kprobe = Arc::new(kprobe);
47    manager.insert_probe(UniProbe::Kprobe(kprobe.clone()));
48    kprobe
49}

51/// Unregister a kprobe.
52///
53/// # Parameters
54/// - `manager`: The kprobe manager.
55/// - `kprobe_point_list`: The list of kprobe points.
56/// - `kprobe`: The kprobe to unregister.
57///
58pub fn unregister_kprobe<L: RawMutex + 'static, F: KprobeAuxiliaryOps>(
59    manager: &mut ProbeManager<L, F>,
60    kprobe_point_list: &mut ProbePointList<F>,
61    kprobe: Arc<Kprobe<L, F>>,
62) {
63    let kprobe_addr = kprobe.probe_point().break_address();
64    manager.remove_probe(UniProbe::Kprobe(kprobe));
65
66    if manager.kprobe_num(kprobe_addr) == 0 {
67        kprobe_point_list.remove(&kprobe_addr);
68    }
69}

71/// Run kprobe which has been registered on the address
72///
73/// # Parameters
74/// - `manager`: The kprobe manager.
75/// - `pt_regs`: The trap pt_regs.
76///
77/// # Returns
78/// - An `Option` containing the result of the kprobe handler. If no kprobe is found, it returns `None`.
79///
80pub fn kprobe_handler_from_break<L: RawMutex + 'static, F: KprobeAuxiliaryOps>(
81    manager: &mut ProbeManager<L, F>,
82    pt_regs: &mut PtRegs,
83) -> Option<()> {
84    let break_addr = pt_regs.break_address();
85    // log::debug!("kprobe_handler_from_break: break_addr: {:#x}", break_addr);
86    let kprobe_list = manager.get_break_list(break_addr);
87    if let Some(kprobe_list) = kprobe_list {
88        for kprobe in kprobe_list {
89            if kprobe.is_enabled() {
90                kprobe.call_pre_handler(pt_regs);
91            }
92        }
93        let single_step_address = kprobe_list[0].probe_point().single_step_address();
94        setup_single_step(pt_regs, single_step_address);
95        Some(())
96    } else {
97        // For some architectures, they do not support single step execution,
98        // and we need to use breakpoint exceptions to simulate
99        kprobe_handler_from_debug(manager, pt_regs)
100    }
101}

103/// Run kprobe which has been registered on the address
104///
105/// # Parameters
106/// - `manager`: The kprobe manager.
107/// - `pt_regs`: The trap pt_regs.
108///
109/// # Returns
110/// - An `Option` containing the result of the kprobe handler. If no kprobe is found, it returns `None`.
111///
112pub fn kprobe_handler_from_debug<L: RawMutex + 'static, F: KprobeAuxiliaryOps>(
113    manager: &mut ProbeManager<L, F>,
114    pt_regs: &mut PtRegs,
115) -> Option<()> {
116    let pc = pt_regs.debug_address();
117    if let Some(kprobe_list) = manager.get_debug_list(pc) {
118        for kprobe in kprobe_list {
119            if kprobe.is_enabled() {
120                kprobe.call_post_handler(pt_regs);
121                kprobe.call_event_callback(pt_regs);
122            }
123        }
124        let return_address = kprobe_list[0].probe_point().return_address();
125        clear_single_step(pt_regs, return_address);
126        Some(())
127    } else {
128        log::info!("There is no kprobe on pc {pc:#x}");
129        None
130    }
