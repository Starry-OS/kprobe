9/// The uprobe structure for the current architecture.
10pub type Uprobe<L, F> = Probe<L, F>;

12/// Register a uprobe.
13///
14/// # Parameters
15/// - `manager`: The uprobe manager.
16/// - `uprobe_point_list`: The list of uprobe points.
17/// - `uprobe_builder`: The uprobe builder.
18///
19/// # Returns
20/// - An registered uprobe.
21///
22pub fn register_uprobe<L: RawMutex + 'static, F: KprobeAuxiliaryOps>(
23    manager: &mut ProbeManager<L, F>,
24    uprobe_point_list: &mut ProbePointList<F>,
25    uprobe_builder: ProbeBuilder<F>,
26) -> Arc<Uprobe<L, F>> {
27    assert!(
28        uprobe_builder.user_pid.is_some(),
29        "The builder must be for uprobe"
30    );
31    let uprobe = crate::__register_kprobe(uprobe_point_list, uprobe_builder);
32    let uprobe = Arc::new(uprobe);
33    manager.insert_probe(UniProbe::Uprobe(uprobe.clone()));
34    uprobe
35}

37/// Unregister a uprobe.
38///
39/// # Parameters
40/// - `manager`: The uprobe manager.
41/// - `uprobe_point_list`: The list of uprobe points.
42/// - `uprobe`: The uprobe to unregister.
43///
44pub fn unregister_uprobe<L: RawMutex + 'static, F: KprobeAuxiliaryOps>(
45    manager: &mut ProbeManager<L, F>,
46    uprobe_point_list: &mut ProbePointList<F>,
47    uprobe: Arc<Uprobe<L, F>>,
48) {
49    let uprobe_addr = uprobe.probe_point().break_address();
50    manager.remove_probe(UniProbe::Uprobe(uprobe));
51
52    if manager.kprobe_num(uprobe_addr) == 0 {
53        uprobe_point_list.remove(&uprobe_addr);
54    }
55}

57/// Run uprobe which has been registered on the address
58///
59/// # Parameters
60/// - `uprobe_manager`: The uprobe manager.
61/// - `pt_regs`: The trap pt_regs.
62///
63/// # Returns
64/// - An `Option` containing the result of the uprobe handler. If no uprobe is found, it returns `None`.
65///
66pub fn uprobe_handler_from_break<L: RawMutex + 'static, F: KprobeAuxiliaryOps>(
67    uprobe_manager: &mut ProbeManager<L, F>,
68    pt_regs: &mut PtRegs,
69) -> Option<()> {
70    let break_addr = pt_regs.break_address();
71    // log::debug!("uprobe_handler_from_break: break_addr: {:#x}", break_addr);
72    let uprobe_list = uprobe_manager.get_break_list(break_addr);
73    if let Some(uprobe_list) = uprobe_list {
74        for uprobe in uprobe_list {
75            if uprobe.is_enabled() {
76                uprobe.call_pre_handler(pt_regs);
77            }
78        }
79
80        let point = uprobe_list[0].probe_point();
81        let dynamic_user_ptr = point.dynamic_user_ptr();
82        if dynamic_user_ptr != 0 {
83            setup_single_step(pt_regs, dynamic_user_ptr);
84        } else {
85            let old_instruction_len = point.old_instruction_len();
86            let single_step_address = point.single_step_address() as *mut u8;
87            let pid = point.pid();
88            let user_ptr = F::alloc_user_exec_memory(pid, |ptr| {
89                unsafe {
90                    // Copy the old instruction back to user space
91                    core::ptr::copy_nonoverlapping(single_step_address, ptr, old_instruction_len);
92                }
93            });
94
95            // the original debug address
96            let old_debug_address = point.debug_address();
97
98            // after this, point.debug_address() will return the new debug address(user_ptr+ debug instruction len)
99            let new_debug_address = point.set_dynamic_user_ptr(user_ptr as usize);
100
101            // beacause the debug address of uprobe is dynamically created, so the key value of the debug_list maintained in the original uprobe_manager needs to be updated
102            uprobe_manager
103                .replace_debug_list_with_new_address(old_debug_address, new_debug_address);
104
105            // update the single step address to the copied instruction
106            setup_single_step(pt_regs, user_ptr as usize);
107        }
108
109        Some(())
110    } else {
111        // For some architectures, they do not support single step execution,
112        // and we need to use breakpoint exceptions to simulate
113        uprobe_handler_from_debug(uprobe_manager, pt_regs)
114    }
115}

117/// Run uprobe which has been registered on the address
118///
119/// # Parameters
120/// - `uprobe_manager`: The uprobe manager.
121/// - `pt_regs`: The trap pt_regs.
122///
123/// # Returns
124/// - An `Option` containing the result of the uprobe handler. If no uprobe is found, it returns `None`.
125///
126pub fn uprobe_handler_from_debug<L: RawMutex + 'static, F: KprobeAuxiliaryOps>(
127    uprobe_manager: &mut ProbeManager<L, F>,
128    pt_regs: &mut PtRegs,
129) -> Option<()> {
130    let pc = pt_regs.debug_address();
131    if let Some(uprobe_list) = uprobe_manager.get_debug_list(pc) {
132        for uprobe in uprobe_list {
133            if uprobe.is_enabled() {
134                uprobe.call_post_handler(pt_regs);
135                uprobe.call_event_callback(pt_regs);
136            }
137        }
138        let return_address = uprobe_list[0].probe_point().return_address();
139        clear_single_step(pt_regs, return_address);
140        Some(())
141    } else {
142        log::debug!("There is no uprobe on pc {pc:#x}");
143        None
144    }
145}
